window.saveBooking = async function() { const docId = document.getElementById('secretaryDoctorSelect').value; const date = document.getElementById('resDate').value; const seq = document.getElementById('resSeq').value.trim(); const rawTime = document.getElementById('resTime').value; const time = formatTimeTo12Hour(rawTime); const name = document.getElementById('resName').value.trim(); const phone = document.getElementById('resPhone').value.trim(); return; } try { // 🛡️ التحقق من عدم تكرار رقم التسلسل لنفس التاريخ والطبيب const dateSnapshot = await get(ref(db, `Bookings/${CLINIC_CODE}/${docId}/${date}`)); if (dateSnapshot.exists()) { const bookingsData = dateSnapshot.val(); for (let key in bookingsData) { // إذا كان التسلسل موجوداً مسبقاً ولا يوافق نفس مفتاح التعديل الحالي if (String(bookingsData[key].sequence) === String(seq) && key !== editKey) { alert(`❌ خطأ: رقم التسلسل (${seq}) محجوز مسبقاً لهذا التاريخ! يرجى اختيار تسلسل آخر.`); return; // إيقاف عملية الحفظ أو التعديل فوراً } } } const bookingData = { sequence: seq, time: time, patientName: name, phone: phone, patientPhone: phone, patientNotes: notes, patientStatus: "قيد الانتظار", timestamp: Date.now() }; if (editKey) { await update(ref(db, `Bookings/${CLINIC_CODE}/${docId}/${date}/${editKey}`), bookingData); alert("✅ تم تحديث الحجز بنجاح!"); cancelEditMode(); } else { await push(ref(db, `Bookings/${CLINIC_CODE}/${docId}/${date}`), bookingData); alert("✅ تم حفظ الحجز بنجاح!"); document.getElementById('resName').value = ''; document.getElementById('resPhone').value = ''; document.getElementById('resNotes').value = ''; document.getElementById('resSeq').value = parseInt(seq) + 1;
